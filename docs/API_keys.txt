pub const REQUIRED_ACCOUNT_KEYS: &[&str] = &[\"solend.USDC.reserve\",\"raydium_clmm.POOL123.vault_a\", ...];
pub fn report_missing_accounts(registry: &RegistryLoaded) {
    let missing: Vec<&str> = REQUIRED_ACCOUNT_KEYS.iter().copied()
       .filter(|k| !registry.has(k))
       .collect();
    if !missing.is_empty() {
        log::warn!(\"account_registry_missing count={} list={:?}\", missing.len(), missing);
        METRICS.account_sets_missing.set(missing.len() as i64);
    }
}
